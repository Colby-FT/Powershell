# Revoke Sign-In Sessions for an array of users
Connect-MgGraph -Scopes "User.ReadWrite.All"

# Configuration
#Fill in the users' UPNs here
#Example: $users = @("user1@contoso.com", "user2@contoso.com")
$users = @()
$dryRun = $true
$logPath = "C:\FT\SessionRevokeLog.txt"

# Do the thing
foreach ($userUPN in $users) {
    try {
        $userObj = Get-MgUser -UserId $userUPN
        $userName = $userObj.DisplayName

        if ($dryRun) {
            Write-Host "[Dry Run] Would revoke all sign in sessions for user $userName with UPN $userUPN"
            Add-Content -Path $logPath -Value "[Dry Run] Would set password for user $userName with UPN $userUPN"
        } 
        else {
            Revoke-MgUserSignInSession -UserId $userUPN
            Write-Host -ForegroundColor Green "Sign in sessions revoked for user $userName with UPN $userUPN"
            Add-Content -Path $logPath -Value "Sign in sessions revoked for user $userName with UPN $userUPN"
        }
    } 
    catch {
        Write-Host -ForegroundColor Red "Failed to process user $userUPN. Error $_"
        Add-Content -Path $logPath -Value "Failed to process user $userUPN. Error $_"
    }
}

#Set the same password for an array of users
Connect-MgGraph -Scopes "User.ReadWrite.All"

# Configuration
$users = @("fttest1@stpaulmn.onmicrosoft.com")
$newPassword = "Cuyo5459"
$dryRun = $true
$logPath = "C:\Path\To\Your\PasswordLog.txt"

# Do the thing
foreach ($userUPN in $users) {
    $userObj = Get-MgUser -UserId $userUPN
    $userName = $userObj.DisplayName
    $params = @{
        PasswordProfile = @{
            Password = $newPassword
            ForceChangePasswordNextSignIn = $true
        }
    }

    if ($dryRun) {
        Write-Host "[Dry Run] Would set password for user $userName with UPN $userUPN"
        Add-Content -Path $logPath -Value "[Dry Run] Would set password for user $userName with UPN $userUPN"
    } else {
        try {
            Update-MgUser -UserId $userUPN -BodyParameter $params
            Write-Host -ForegroundColor Green "Password set successfully for user $userName with UPN $userUPN"
            Add-Content -Path $logPath -Value "Password set successfully for user $userName with UPN $userUPN"
        } catch {
            Write-Host -ForegroundColor Red "Failed to set password for user $userName with UPN $userUPN. Error: $_"
            Add-Content -Path $logPath -Value "Failed to set password for user $userName with UPN $userUPN. Error: $_"
        }
    }
}

##Get any users not in an MFA enforced group if using Conditional Access
#Get All Conditional Access Policies
Connect-MgGraph -Scopes "Policy.Read.All", "Group.Read.All"

$policies = Get-MgIdentityConditionalAccessPolicy | Where-Object {
    $_.Conditions -and $_.GrantControls.BuiltInControls -contains "mfa"
}

#Extract the groups from the policies
$groupIds = $policies.Conditions.Users.IncludeGroups

#Get the members of the groups
$enforcedUsers = foreach ($groupId in $groupIds) {
    Get-MgGroupMember -GroupId $groupId -All | Where-Object {
        $_.AdditionalProperties.'@odata.type' -eq '#microsoft.graph.user'
    } | Select-Object -ExpandProperty Id
}

#Get all users in the tenant
$allUsers = Get-MgUser -All | Select-Object Id, UserPrincipalName

#Compare the lists to find users not in any enforced group
$unenforcedUsers = $allUsers | Where-Object {
    $enforcedUsers -notcontains $_.Id
}

$unenforcedUsers | Select-Object UserPrincipalName

#Export to CSV
$unenforcedUsers | Export-Csv -Path "C:\FT\SPC-UsersWithoutMFAEnforcement.csv" -NoTypeInformation

#Remove all authentication methods for an array of users
# Connect to Microsoft Graph

# Configuration

$logPath = "C:\Path\To\Your\Log.txt"            # Path to your log file

foreach ($userUPN in $users) {
    $authMethods = Get-MgUserAuthenticationMethod -UserId $userUPN
    $userObj = Get-MgUser -UserId $userUPN
    $userName = $userObj.DisplayName

    foreach ($method in $authMethods) {
        $id = $method.Id
        $type = $method.AdditionalProperties["@odata.type"]
