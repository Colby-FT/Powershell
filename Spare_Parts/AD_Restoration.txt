#Map replication between each DC
function Invoke-ADSchemaReplicationAllDCs {
    # Get all domain controllers in the domain
    $allDCs = Get-ADDomainController -Filter * | Select-Object -ExpandProperty HostName

    # Get the schema naming context
    $schemaDN = (Get-ADRootDSE).schemaNamingContext

    foreach ($sourceDC in $allDCs) {
        $targetDCs = $allDCs | Where-Object { $_ -ne $sourceDC }
        foreach ($target in $targetDCs) {
            Sync-ADObject -Object (Get-ADObject -Identity $schemaDN) -Source $sourceDC -Destination $target
        }
    }
}

Invoke-ADSchemaReplicationAllDCs

===================Check initial replication status===================
# Define source and new DC names
$SourceDC = "shvusdc001"   # Replace with an existing DC
$NewDC    = $env:COMPUTERNAME  # Current DC

Write-Host "Checking SYSVOL and NTDS replication status on $NewDC..." -ForegroundColor Cyan

# 1. Check DFSR Replication State
$dfsrState = dfsrdiag ReplicationState 2>$null
if ($dfsrState) {
    Write-Host "`n[DFSR] Replication in progress:" -ForegroundColor Yellow
    $dfsrState
} else {
    Write-Host "`n[DFSR] No active replication detected." -ForegroundColor Green
}

# 2. Check DFSR Backlog
Write-Host "`nChecking DFSR backlog for SYSVOL..."
$backlog = dfsrdiag backlog /rgname:"Domain System Volume" /rfname:"SYSVOL Share" /sendingmember:$SourceDC /receivingmember:$NewDC 2>$null
if ($backlog) {
    Write-Host "[DFSR] Backlog detected:" -ForegroundColor Yellow
    $backlog
} else {
    Write-Host "[DFSR] No backlog detected (SYSVOL likely in sync)." -ForegroundColor Green
}

# 3. Check SYSVOL and NETLOGON shares
Write-Host "`nChecking SYSVOL and NETLOGON shares..."
$shares = net share
if ($shares -match "SYSVOL" -and $shares -match "NETLOGON") {
    Write-Host "[Shares] SYSVOL and NETLOGON are shared." -ForegroundColor Green
} else {
    Write-Host "[Shares] SYSVOL/NETLOGON not shared yet." -ForegroundColor Red
}

# 4. Check NTDS replication status
Write-Host "`nChecking NTDS replication status..."
$ntdsStatus = repadmin /showrepl $NewDC /errorsonly
if ($ntdsStatus) {
    Write-Host "[NTDS] Replication errors detected:" -ForegroundColor Red
    $ntdsStatus
} else {
    Write-Host "[NTDS] No replication errors detected." -ForegroundColor Green
}

Write-Host "`nSummary:"
Write-Host "✔ DFSR active replication: " ($dfsrState -ne $null)
Write-Host "✔ DFSR backlog: " ($backlog -ne $null)
Write-Host "✔ SYSVOL shared: " ($shares -match "SYSVOL")
Write-Host "✔ NTDS replication healthy: " ($ntdsStatus -eq $null)
