###################
##NetSH DHCP
#Export DHCP settings
netsh dhcp server export C:\FT\dhcp-$hname.txt all

#Import DHCP settings
netsh dhcp server import C:\FT\dhxp-<name>.txt all

##################
##Backup DHCP
#Export DHCP Settings
Backup-dhcpserver -path C:\FT\DHCPBackup

#Import DHCP Settings
#Remember to place the "DHCPBackup" file from the last step into the folder "C:\Windows\system32\dhcp\backup"
Restore-dhcspserver -computername "<PCName>" -Path "C:\Windows\system32\dhcp\backup"

#################
##Demote the DC
# Import the ADDSDeployment module
Import-Module ADDSDeployment

# Uninstall the AD DS role and demote the domain controller
Uninstall-ADDSDomainController -DemoteOperationMasterRole:$true -RemoveDnsDelegation:$true -Force:$true

# Uninstall the AD DS management tools
Uninstall-WindowsFeature AD-Domain-Services -IncludeManagementTools

#####################
#Repair Domain Trust
Test-ComputerSecureChannel -Repair -Server <PDCHere> -Credential "<UsernameHere>"
# Example: Test-ComputerSecureChannel -Repair -Server "IKAV-FRP-DC01" -Credential "FTadmin"

#########################
##Promote to a DC
#Join the domain
add-computer -ComputerName NDI-WDC-002 -DomainName VW-DOMAIN -Server NDI-WDC-03

#Install AD DS Management tools
Install-WindowsFeature AD-Domain-services, DNS -IncludeManagementTools

#Promote to DC
